% D_t = akumulasi panjang sebelum langkah; d_t = jarak lurus sebelum langkah.
% Terminal langsung = success, over_distance, atau step_limit dari step.
\Delta d_t=d_t-d_{t+1}
R_{\mathrm{arah},t}=
\begin{cases}
0, & \text{terminal langsung},\\
-0{,}60\frac{\Delta d_t}{100}, & D_t<2500\ \text{dan}\ \Delta d_t>0,\\
+0{,}60\frac{\Delta d_t}{100}, & D_{t+1}\ge2500\ \text{dan kondisi sebelumnya tidak berlaku},\\
0, & \text{lainnya}.
\end{cases}

R_t=R_{\mathrm{ruas},t}+R_{\mathrm{akses},t}+R_{\mathrm{arah},t}
    +R_{\mathrm{kunjungan},t}+R_{\mathrm{terminal},t}

% Metrik, penjumlahan atas setiap perpindahan i=1,...,n

\mathrm{absolute\_distance\_error\_m}=|D-5000|
\mathrm{return\_progress\_ratio}=\frac{N_{\mathrm{mendekat}}}{N_{\mathrm{pulang}}}
% Fase metrik: akumulasi setelah langkah >=2500 m; kosong bila tidak ada langkah fase ini.
\mathrm{mean\_comfort}=\frac{\sum_{i=1}^{n}C_{e_i}L_{e_i}}{\sum_{i=1}^{n}L_{e_i}}
\mathrm{overlap\_ratio}=\frac{N_{\mathrm{lintasan}}-N_{\mathrm{ruas\ unik}}}{N_{\mathrm{lintasan}}}
\mathrm{success\ rate}=\frac{\mathrm{jumlah\ seed\ dengan\ is\_loop=True}}{5}\times100\%
\mathrm{termination\ rate}_j=\frac{\mathrm{jumlah\ seed\ dengan\ terminasi}\ j}{5}\times100\%
